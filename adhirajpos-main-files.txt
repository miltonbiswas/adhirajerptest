/**
 * HSN Master
 *
 * Existing HSN records are loaded from the backend.
 * Only active HSN codes are offered for selection.
 */
async function loadHsn(force = false) {
  if (hsnLoading.value) return
  if (hsnLoaded.value && !force) return

  hsnLoading.value = true
  hsnError.value = ''

  try {
    const response = await api.getHsn({
      active: true,
      page: 1,
      limit: 500,
    })

    hsnRecords.value = response.data
    hsnLoaded.value = true
  } catch (error: unknown) {
    hsnError.value = getErrorMessage(
      error,
      'Unable to load HSN master records.',
    )
  } finally {
    hsnLoading.value = false
  }
}

}

    popupError.value = 'Please enter a name.'
    return

    emit('add-composition')
  }

  closeAddPopup()
}

function getErrorMessage(error: unknown, fallback: string): string {
  if (
    error &&
    typeof error === 'object' &&
    'data' in error
  ) {
    const data = (error as { data?: { message?: string; error?: string } }).data

    if (data?.message) return data.message
    if (data?.error) return data.error
  }

  if (error instanceof Error && error.message) {
    return error.message
  }

  return fallback
}

/**
 * Creates an HSN record through the existing HSN API.
 *
 * The GST split is calculated as CGST + SGST for the
 * entered total GST rate. This assumes an intra-state
 * transaction; the actual invoice tax treatment should
 * be handled by the billing/tax logic.
 */
async function submitNewHsn() {
  const code = newHsnCode.value.trim()
  const description = newHsnDescription.value.trim()
  const gstRate = Number(newHsnGstRate.value)

  hsnError.value = ''

  if (!code) {
    hsnError.value = 'Please enter an HSN code.'
    return
  }

  if (!/^\d{4,8}$/.test(code)) {
    hsnError.value = 'Enter a valid 4â€“8 digit HSN code.'
    return
  }

  if (!Number.isFinite(gstRate) || gstRate < 0 || gstRate > 100) {
    hsnError.value = 'GST rate must be between 0 and 100.'
    return
  }

  const duplicate = hsnRecords.value.find(
    item => item.hsn_code === code,
  )

  if (duplicate) {
    hsnError.value = 'This HSN code already exists in HSN Master.'

    if (duplicate.is_active) {
      selectHsn(duplicate)
      closeAddPopup()
    }

    return
  }

  const payload: HsnInput = {
    hsn_code: code,
    description,
    gst_rate: gstRate,
    cgst_rate: gstRate / 2,
    sgst_rate: gstRate / 2,
    igst_rate: gstRate,
  }

  hsnSaving.value = true

  try {
    const created = await api.createHsn(payload)

    await loadHsn(true)

    const record =
      hsnRecords.value.find(
        item => item.hsn_code === created.hsn_code,
      ) || created

    selectHsn(record)
    closeAddPopup()
  } catch (error: unknown) {
    hsnError.value = getErrorMessage(
      error,
      'Unable to create HSN record.',
    )
  } finally {
    hsnSaving.value = false
  }
}

function onDocumentClick(event: MouseEvent) {
  if (
    dropdownRoot.value &&
    !dropdownRoot.value.contains(event.target as Node)
  ) {
    activeDropdown.value = null
  }
}

function onKeydown(event: KeyboardEvent) {
  if (event.key === 'Escape') {
    if (activePopup.value) {
      closeAddPopup()
    } else if (activeDropdown.value) {
      activeDropdown.value = null
    } else {
      emit('close')
    }
  }
}

onMounted(() => {

})

onBeforeUnmount(() => {
  document.removeEventListener('click', onDocumentClick)
  document.removeEventListener('keydown', onKeydown)
})

function expiryState(value: string | null | undefined): 'expired' | 'soon' | 'valid' | 'unknown' {
  if (!value) return 'unknown'

  const expiry = new Date(`${value.slice(0, 10)}T00:00:00`)
  if (Number.isNaN(expiry.getTime())) return 'unknown'

  const today = new Date()
  today.setHours(0, 0, 0, 0)

  const days = Math.ceil((expiry.getTime() - today.getTime()) / 86400000)

  if (days < 0) return 'expired'
  if (days <= 90) return 'soon'
  return 'valid'
}

async function submitAdjustment() {
  actionError.value = ''
  actionSuccess.value = ''

  if (!adjustment.value.reason.trim()) {
    actionError.value = 'Please enter a reason for this stock movement.'
    return
  }

  if (!Number.isFinite(Number(adjustment.value.quantity)) || Number(adjustment.value.quantity) < 0) {
    actionError.value = 'Enter a valid quantity.'
    return
  }

  if (adjustment.value.movement_type !== 'ADJUSTMENT' && Number(adjustment.value.quantity) <= 0) {
    actionError.value = 'Quantity must be greater than zero.'
    return
  }

  if (selectedBatch.value && adjustment.value.movement_type === 'OUT') {
    if (Number(adjustment.value.quantity) > Number(selectedBatch.value.quantity)) {
      actionError.value = 'Quantity cannot exceed the available stock.'
      return
    }
  }

  // Creating stock for an existing product requires a batch record first.
  // Do not fabricate a batch or expiry date.
  if (!selectedBatch.value && selectedProduct.value) {
    actionError.value = 'This product has no batch record. Add a batch through the purchase/batch workflow before adjusting stock.'
    return
  }

  if (!selectedBatch.value) {
    actionError.value = 'Select a batch before adjusting stock.'
    return
  }

  actionLoading.value = true

  try {
    await request('/inventory/adjustments', {
      method: 'POST',
      body: {
        product_id: selectedBatch.value.product_id,
        batch_id: selectedBatch.value.id,
        movement_type: adjustment.value.movement_type,
        quantity: Number(adjustment.value.quantity),
        reason: adjustment.value.reason.trim(),
      },
    })

    showAdjustment.value = false
    actionSuccess.value = 'Stock adjustment saved.'
    await refreshAll()
  } catch (error: unknown) {
    actionError.value = getError(error)
  } finally {
    actionLoading.value = false
  }
}

<script setup lang="ts">
import { computed, onBeforeUnmount, onMounted, ref, watch } from 'vue'
import type { Party } from '~/composables/useApi'
import PartyFormModal from '~/components/parties/PartyFormModal.vue'

useSeoMeta({ title: 'Suppliers | AdhirajPOS' })

const { getParties, deactivateParty } = useApi()

const suppliers = ref<Party[]>([])
const loading = ref(false)
const actionLoading = ref<string | null>(null)
const pageError = ref('')
const successMessage = ref('')
const search = ref('')
const activeFilter = ref<'all' | 'active' | 'inactive'>('active')
const page = ref(1)
const pageSize = 25
const total = ref(0)

const modalOpen = ref(false)
const selectedParty = ref<Party | null>(null)

let searchTimer: ReturnType<typeof setTimeout> | undefined
let messageTimer: ReturnType<typeof setTimeout> | undefined
let requestSequence = 0

const totalPages = computed(() =>
  Math.max(1, Math.ceil(total.value / pageSize)),
)

const startRecord = computed(() =>
  total.value > 0 ? (page.value - 1) * pageSize + 1 : 0,
)

const endRecord = computed(() =>
  Math.min(page.value * pageSize, total.value),
)

const formatCurrency = (value: number | string | null | undefined) => {
  const amount = Number(value ?? 0)

  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    maximumFractionDigits: 2,
  }).format(Number.isFinite(amount) ? amount : 0)
}

const formatLocation = (party: Party) =>
  [party.city, party.state].filter(Boolean).join(', ') || 'â€”'

const formatBalanceType = (party: Party) => {
  switch (party.opening_balance_type) {
    case 'PAYABLE':
      return 'Payable'
    case 'RECEIVABLE':
      return 'Receivable'
    case 'ADVANCE':
      return 'Advance'
    default:
      return 'Opening balance'
  }
}

function showMessage(message: string, type: 'success' | 'error') {
  if (messageTimer) clearTimeout(messageTimer)

  if (type === 'success') {
    successMessage.value = message
    pageError.value = ''
  } else {
    pageError.value = message
    successMessage.value = ''
  }

  messageTimer = setTimeout(() => {
    successMessage.value = ''
    pageError.value = ''
  }, 5000)
}

async function loadSuppliers() {
  const sequence = ++requestSequence

  loading.value = true
  pageError.value = ''

  try {
    const response = await getParties({
      search: search.value.trim() || undefined,
      role: 'SUPPLIER',
      page: page.value,
      limit: pageSize,
      active:
        activeFilter.value === 'all'
          ? undefined
          : activeFilter.value === 'active',
    })

    if (sequence !== requestSequence) return

    suppliers.value = response.data ?? []
    total.value = Number(response.total ?? 0)

    const lastPage = Math.max(1, Math.ceil(total.value / pageSize))

    if (page.value > lastPage) {
      page.value = lastPage
    }
  } catch (error: unknown) {
    if (sequence !== requestSequence) return

    suppliers.value = []
    total.value = 0

    pageError.value =
      error instanceof Error
        ? error.message
        : 'Unable to load suppliers. Please try again.'
  } finally {
    if (sequence === requestSequence) {
      loading.value = false
    }
  }
}

function handleSearch() {
  if (searchTimer) clearTimeout(searchTimer)

  searchTimer = setTimeout(() => {
    if (page.value !== 1) {
      page.value = 1
    } else {
      void loadSuppliers()
    }
  }, 300)
}

function resetFilters() {
  if (searchTimer) clearTimeout(searchTimer)

  search.value = ''
  activeFilter.value = 'active'
  page.value = 1

  void loadSuppliers()
}

function openCreateModal() {
  selectedParty.value = null
  modalOpen.value = true
}

function openEditModal(party: Party) {
  selectedParty.value = party
  modalOpen.value = true
}

async function handleSaved() {
  modalOpen.value = false
  selectedParty.value = null

  showMessage('Supplier saved successfully.', 'success')
  await loadSuppliers()
}

async function handleDeactivate(party: Party) {
  if (!party.is_active || actionLoading.value) return

  const confirmed = window.confirm(
    `Deactivate "${party.name}"?\n\nThe supplier will remain in your records but will no longer appear in the active directory.`,
  )

  if (!confirmed) return

  actionLoading.value = party.id
  pageError.value = ''
  successMessage.value = ''

  try {
    await deactivateParty(party.id)

    showMessage('Supplier deactivated successfully.', 'success')

    if (suppliers.value.length === 1 && page.value > 1) {
      page.value -= 1
    } else {
      await loadSuppliers()
    }
  } catch (error: unknown) {
    showMessage(
      error instanceof Error
        ? error.message
        : 'Unable to deactivate this supplier.',
      'error',
    )
  } finally {
    actionLoading.value = null
  }
}

function changePage(nextPage: number) {
  if (nextPage < 1 || nextPage > totalPages.value || loading.value) return
  page.value = nextPage
}

watch([activeFilter, page], () => {
  void loadSuppliers()
})

onMounted(() => {
  void loadSuppliers()
})

onBeforeUnmount(() => {
  if (searchTimer) clearTimeout(searchTimer)
  if (messageTimer) clearTimeout(messageTimer)
  requestSequence++
})
</script>

use anyhow::{Context, Result};
use std::env;

#[derive(Debug, Clone)]
pub struct Config {
    pub db_host: String,
    pub db_port: u16,
    pub db_name: String,
    pub db_user: String,
    pub db_password: String,
    pub server_host: String,
    pub server_port: u16,
}

impl Config {
    pub fn from_env() -> Result<Self> {
        dotenvy::dotenv().ok();

        let db_host = required_var("DB_HOST")?;
        let db_port = required_var("DB_PORT")?
            .parse::<u16>()
            .context("DB_PORT must be a valid port number")?;
        let db_name = required_var("DB_NAME")?;
        let db_user = required_var("DB_USER")?;
        let db_password = required_var("DB_PASSWORD")?;

        let server_host = env::var("SERVER_HOST").unwrap_or_else(|_| "127.0.0.1".to_string());

        let server_port = env::var("SERVER_PORT")
            .unwrap_or_else(|_| "8080".to_string())
            .parse::<u16>()
            .context("SERVER_PORT must be a valid port number")?;

        Ok(Self {
            db_host,
            db_port,
            db_name,
            db_user,
            db_password,
            server_host,
            server_port,
        })
    }
}

fn required_var(name: &str) -> Result<String> {
    let value = env::var(name)
        .with_context(|| format!("Required environment variable {name} is missing"))?;

    if value.trim().is_empty() {
        anyhow::bail!("Environment variable {name} cannot be empty");
    }

    Ok(value)
}

use anyhow::{Context, Result};
use sqlx::postgres::PgPoolOptions;
use sqlx::{PgPool, postgres::PgConnectOptions};
use std::time::Duration;

use crate::config::Config;

pub async fn create_pool(config: &Config) -> Result<PgPool> {
    let options = PgConnectOptions::new()
        .host(&config.db_host)
        .port(config.db_port)
        .database(&config.db_name)
        .username(&config.db_user)
        .password(&config.db_password);

    let pool = PgPoolOptions::new()
        .max_connections(10)
        .min_connections(1)
        .acquire_timeout(Duration::from_secs(10))
        .connect_with(options)
        .await
        .context("Failed to connect to PostgreSQL")?;

    sqlx::query("SELECT 1")
        .execute(&pool)
        .await
        .context("PostgreSQL connection test failed")?;

    // Apply pending database migrations.
    sqlx::migrate!("./migrations")
        .run(&pool)
        .await
        .context("Failed to apply database migrations")?;

    tracing::info!("Database connected and migrations applied successfully.");

    Ok(pool)
}

pub async fn save_business_profile(
    axum::extract::State(pool): axum::extract::State<PgPool>,

    let business_name = normalize(&input.business_name);

    if business_name.is_empty() {
        return error_response(StatusCode::BAD_REQUEST, "Business name is required")
            .into_response();
    }

    if business_name.chars().count() > 200 {
        return error_response(
            StatusCode::BAD_REQUEST,
            "Business name must not exceed 200 characters",
        )
        .into_response();
    }

    let pincode = normalize(&input.pincode);

    if !pincode.is_empty() && (pincode.len() != 6 || !pincode.chars().all(|c| c.is_ascii_digit())) {
        return error_response(
            StatusCode::BAD_REQUEST,
            "PIN code must contain exactly 6 digits",
        )
        .into_response();
    }

    let gstin = normalize(&input.gstin).to_ascii_uppercase();

    if !gstin.is_empty() && (gstin.len() != 15 || !gstin.chars().all(|c| c.is_ascii_alphanumeric()))
    {
        return error_response(
            StatusCode::BAD_REQUEST,
            "GSTIN must contain 15 alphanumeric characters",
        )
        .into_response();

// ============================================================
// CSV BULK IMPORT
// POST /api/hsn/import
//
// Multipart form field: file
// Required CSV headers:
// hsn_code,gst_rate,cgst_rate,sgst_rate,igst_rate
//
// Existing records are never overwritten.
// Invalid rows are reported.
// Database errors roll back the transaction.
// ============================================================

pub async fn import_hsn_csv(
    State(pool): State<PgPool>,
    mut multipart: Multipart,
) -> Result<Json<HsnImportResult>, (StatusCode, Json<ApiError>)> {
    let mut csv_bytes: Option<Vec<u8>> = None;

    while let Some(field) = multipart
        .next_field()
        .await
        .map_err(|_| bad_request("Could not read uploaded file"))?
    {
        if field.name() == Some("file") {
            let bytes = field
                .bytes()
                .await
                .map_err(|_| bad_request("Could not read CSV file"))?;

            if bytes.len() > 5 * 1024 * 1024 {
                return Err(bad_request("CSV file must be 5 MB or smaller"));
            }

            csv_bytes = Some(bytes.to_vec());
            break;
        }
    }

    let bytes =
        csv_bytes.ok_or_else(|| bad_request("Upload a CSV using the multipart field 'file'"))?;

    if bytes.is_empty() {
        return Err(bad_request("CSV file is empty"));
    }

    let mut reader = csv::ReaderBuilder::new()
        .trim(csv::Trim::All)
        .from_reader(bytes.as_slice());

    let headers = reader
        .headers()
        .map_err(|_| bad_request("Could not parse CSV headers"))?
        .clone();

    let required_headers = [
        "hsn_code",
        "gst_rate",
        "cgst_rate",
        "sgst_rate",
        "igst_rate",
    ];

    for required in required_headers {
        if !headers.iter().any(|header| header == required) {
            return Err(bad_request(&format!(
                "CSV is missing required column: {required}"
            )));
        }
    }

    let mut result = HsnImportResult {
        total_rows: 0,
        inserted: 0,
        skipped: 0,
        rejected: 0,
        errors: Vec::new(),
    };

    let mut tx = pool.begin().await.map_err(internal_error)?;

    for (index, record) in reader.deserialize::<HsnCsvRow>().enumerate() {
        result.total_rows += 1;
        let row_number = index + 2;

        let row = match record {
            Ok(row) => row,
            Err(error) => {
                result.rejected += 1;
                result.errors.push(HsnImportError {
                    row: row_number,
                    hsn_code: String::new(),
                    error: format!("Invalid CSV row: {error}"),
                });
                continue;
            }
        };

        let code = row.hsn_code.trim();

        if !is_valid_hsn_code(code) {
            result.rejected += 1;
            result.errors.push(HsnImportError {
                row: row_number,
                hsn_code: code.to_owned(),
                error: "HSN code must contain 1-8 ASCII letters, digits or dots".to_owned(),
            });
            continue;
        }

        let rates = [row.gst_rate, row.cgst_rate, row.sgst_rate, row.igst_rate];

        if rates
            .iter()
            .any(|rate| *rate < Decimal::ZERO || *rate > Decimal::from(100))
        {
            result.rejected += 1;
            result.errors.push(HsnImportError {
                row: row_number,
                hsn_code: code.to_owned(),
                error: "Tax rates must be between 0 and 100".to_owned(),
            });
            continue;
        }

        let insert_result = sqlx::query(
            r#"
            INSERT INTO public.hsn_master (
                hsn_code,
                description,
                gst_rate,
                cgst_rate,
                sgst_rate,
                igst_rate
            )
            VALUES ($1, '', $2, $3, $4, $5)
            ON CONFLICT (hsn_code) DO NOTHING
            "#,
        )
        .bind(code)
        .bind(row.gst_rate)
        .bind(row.cgst_rate)
        .bind(row.sgst_rate)
        .bind(row.igst_rate)
        .execute(&mut *tx)
        .await
        .map_err(internal_error)?;

        if insert_result.rows_affected() == 1 {
            result.inserted += 1;
        } else {
            result.skipped += 1;
        }
    }

    tx.commit().await.map_err(internal_error)?;

    Ok(Json(result))
}

// ============================================================
// VALIDATION
// ============================================================

fn is_valid_hsn_code(code: &str) -> bool {
    !code.is_empty()
        && code.len() <= 8
        && code
            .bytes()
            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'.')
}

fn validate_hsn(code: &str, input: &HsnInput) -> Result<(), (StatusCode, Json<ApiError>)> {
    if !is_valid_hsn_code(code) {
        return Err(bad_request(
            "HSN code must contain 1-8 ASCII letters, digits or dots",
        ));
    }

    let rates = [
        input.gst_rate,
        input.cgst_rate,
        input.sgst_rate,
        input.igst_rate,
    ];

    if rates
        .iter()
        .any(|rate| *rate < Decimal::ZERO || *rate > Decimal::from(100))
    {
        return Err(bad_request("Tax rates must be between 0 and 100"));
    }

    Ok(())
}

// ============================================================
// ERROR HELPERS
// ============================================================

fn bad_request(message: &str) -> (StatusCode, Json<ApiError>) {
    (
        StatusCode::BAD_REQUEST,
        Json(ApiError {
            error: message.to_owned(),
        }),
    )
}

fn not_found() -> (StatusCode, Json<ApiError>) {
    (
        StatusCode::NOT_FOUND,
        Json(ApiError {
            error: "HSN code not found".to_owned(),
        }),
    )
}

fn internal_error(error: sqlx::Error) -> (StatusCode, Json<ApiError>) {
    tracing::error!("HSN database operation failed: {error}");

    (
        StatusCode::INTERNAL_SERVER_ERROR,
        Json(ApiError {
            error: "Database operation failed".to_owned(),
        }),
    )
}

fn write_error(error: sqlx::Error) -> (StatusCode, Json<ApiError>) {
    if let sqlx::Error::Database(ref db_error) = error {
        if db_error.is_unique_violation() {
            return (
                StatusCode::CONFLICT,
                Json(ApiError {
                    error: "This HSN code already exists".to_owned(),
                }),
            );
        }

        if db_error.is_check_violation() {
            return (
                StatusCode::BAD_REQUEST,
                Json(ApiError {
                    error: "One or more tax rates are invalid".to_owned(),
                }),
            );
        }
    }

    tracing::error!("HSN write failed: {error}");

    (
        StatusCode::INTERNAL_SERVER_ERROR,
        Json(ApiError {
            error: "HSN operation failed".to_owned(),
        }),
    )
}

#[derive(Debug, Serialize, FromRow)]
pub struct InventorySummary {
    pub total_products: i64,
    pub total_batches: i64,
    pub total_units: Decimal,
    pub low_stock: i64,
    pub out_of_stock: i64,
    pub expiring_soon: i64,
    pub expired: i64,
    pub stock_value: Decimal,
}

pub async fn get_inventory_summary(
    State(pool): State<PgPool>,
) -> Result<Json<InventorySummary>, ApiError> {
    let summary = sqlx::query_as::<_, InventorySummary>(
        r#"
        WITH product_stock AS (
            SELECT
                p.id AS product_id,

                COALESCE(
                    SUM(b.quantity_on_hand)
                    FILTER (WHERE b.is_active = TRUE),
                    0
                ) AS quantity,

                COALESCE(
                    SUM(b.reorder_level)
                    FILTER (WHERE b.is_active = TRUE),
                    0
                ) AS reorder_level

            FROM products p
            LEFT JOIN product_batches b
                ON b.product_id = p.id

            WHERE p.is_active = TRUE
            GROUP BY p.id
        ),

        batch_summary AS (
            SELECT
                COUNT(*) FILTER (
                    WHERE b.is_active = TRUE
                      AND b.quantity_on_hand > 0
                      AND b.expiry_date >= CURRENT_DATE
                      AND b.expiry_date <= CURRENT_DATE + 90
                )::BIGINT AS expiring_soon,

                COUNT(*) FILTER (
                    WHERE b.is_active = TRUE
                      AND b.quantity_on_hand > 0
                      AND b.expiry_date < CURRENT_DATE
                )::BIGINT AS expired,

                COUNT(*) FILTER (
                    WHERE b.is_active = TRUE
                )::BIGINT AS total_batches,

                COALESCE(
                    SUM(b.quantity_on_hand)
                    FILTER (WHERE b.is_active = TRUE),
                    0
                ) AS total_units,

                COALESCE(
                    SUM(b.quantity_on_hand * b.purchase_rate)
                    FILTER (WHERE b.is_active = TRUE),
                    0
                ) AS stock_value

            FROM product_batches b
            JOIN products p
                ON p.id = b.product_id

            WHERE p.is_active = TRUE
        )

        SELECT
            (SELECT COUNT(*) FROM product_stock)::BIGINT
                AS total_products,

            bs.total_batches,
            bs.total_units::NUMERIC AS total_units,

            (
                SELECT COUNT(*)
                FROM product_stock
                WHERE quantity > 0
                  AND quantity <= reorder_level
            )::BIGINT AS low_stock,

            (
                SELECT COUNT(*)
                FROM product_stock
                WHERE quantity = 0
            )::BIGINT AS out_of_stock,

            bs.expiring_soon,
            bs.expired,
            bs.stock_value::NUMERIC AS stock_value

        FROM batch_summary bs
        "#,
    )
    .fetch_one(&pool)
    .await
    .map_err(ApiError::internal)?;

    Ok(Json(summary))
}

pub async fn list_inventory_batches(
    State(pool): State<PgPool>,
    Query(params): Query<BatchQuery>,
) -> Result<Json<BatchListResponse>, ApiError> {
    let page = params.page.unwrap_or(1).max(1);
    let limit = params.limit.unwrap_or(25).clamp(1, 100);
    let offset = (page - 1) * limit;

    let search = params.search.unwrap_or_default().trim().to_owned();
    let pattern = format!("%{search}%");

    let stock_status = params
        .stock_status
        .unwrap_or_else(|| "all".to_owned())
        .to_lowercase();

    let expiry_status = params
        .expiry_status
        .unwrap_or_else(|| "all".to_owned())
        .to_lowercase();

    let valid_stock_status = ["all", "in_stock", "low_stock", "out_of_stock"];
    let valid_expiry_status = ["all", "expired", "expiring_30", "expiring_90", "valid"];

    if !valid_stock_status.contains(&stock_status.as_str()) {
        return Err(ApiError::bad_request("Invalid stock_status filter"));
    }

    if !valid_expiry_status.contains(&expiry_status.as_str()) {
        return Err(ApiError::bad_request("Invalid expiry_status filter"));
    }

    let total: i64 = sqlx::query_scalar(
        r#"
        SELECT COUNT(*)
        FROM product_batches b
        JOIN products p ON p.id = b.product_id

        WHERE p.is_active = TRUE
          AND b.is_active = TRUE

          AND (
              $1 = ''
              OR p.name ILIKE $2
              OR p.sku ILIKE $2
              OR b.batch_number ILIKE $2
              OR COALESCE(p.brand_name, '') ILIKE $2
              OR COALESCE(p.generic_name, '') ILIKE $2
          )

          AND (
              $3 = 'all'
              OR ($3 = 'in_stock' AND b.quantity_on_hand > b.reorder_level)
              OR ($3 = 'low_stock'
                  AND b.quantity_on_hand > 0
                  AND b.quantity_on_hand <= b.reorder_level)
              OR ($3 = 'out_of_stock' AND b.quantity_on_hand = 0)
          )

          AND (
              $4 = 'all'
              OR ($4 = 'expired' AND b.expiry_date < CURRENT_DATE)
              OR ($4 = 'expiring_30'
                  AND b.expiry_date >= CURRENT_DATE
                  AND b.expiry_date <= CURRENT_DATE + 30)
              OR ($4 = 'expiring_90'
                  AND b.expiry_date >= CURRENT_DATE
                  AND b.expiry_date <= CURRENT_DATE + 90)
              OR ($4 = 'valid'
                  AND (b.expiry_date IS NULL OR b.expiry_date > CURRENT_DATE + 90))
          )
        "#,
    )
    .bind(&search)
    .bind(&pattern)
    .bind(&stock_status)
    .bind(&expiry_status)
    .fetch_one(&pool)
    .await
    .map_err(ApiError::internal)?;

    let data = sqlx::query_as::<_, InventoryBatch>(
        r#"
        SELECT
            b.id,
            p.id AS product_id,
            p.name AS product_name,
            p.sku,
            p.brand_name,
            b.batch_number,
            b.manufacturing_date,
            b.expiry_date,
            b.quantity_on_hand AS quantity,
            b.reorder_level,
            p.unit,
            b.purchase_rate,
            b.mrp,
            b.selling_rate

        FROM product_batches b
        JOIN products p
            ON p.id = b.product_id

        WHERE p.is_active = TRUE
          AND b.is_active = TRUE

          AND (
              $1 = ''
              OR p.name ILIKE $2
              OR p.sku ILIKE $2
              OR b.batch_number ILIKE $2
              OR COALESCE(p.brand_name, '') ILIKE $2
              OR COALESCE(p.generic_name, '') ILIKE $2
          )

          AND (
              $3 = 'all'
              OR ($3 = 'in_stock' AND b.quantity_on_hand > b.reorder_level)
              OR ($3 = 'low_stock'
                  AND b.quantity_on_hand > 0
                  AND b.quantity_on_hand <= b.reorder_level)
              OR ($3 = 'out_of_stock' AND b.quantity_on_hand = 0)
          )

          AND (
              $4 = 'all'
              OR ($4 = 'expired' AND b.expiry_date < CURRENT_DATE)
              OR ($4 = 'expiring_30'
                  AND b.expiry_date >= CURRENT_DATE
                  AND b.expiry_date <= CURRENT_DATE + 30)
              OR ($4 = 'expiring_90'
                  AND b.expiry_date >= CURRENT_DATE
                  AND b.expiry_date <= CURRENT_DATE + 90)
              OR ($4 = 'valid'
                  AND (b.expiry_date IS NULL OR b.expiry_date > CURRENT_DATE + 90))
          )

        ORDER BY
            b.expiry_date ASC NULLS LAST,
            p.name ASC,
            b.batch_number ASC

        LIMIT $5 OFFSET $6
        "#,
    )
    .bind(&search)
    .bind(&pattern)
    .bind(&stock_status)
    .bind(&expiry_status)
    .bind(limit)
    .bind(offset)
    .fetch_all(&pool)
    .await
    .map_err(ApiError::internal)?;

    Ok(Json(BatchListResponse {

        page,
        limit,
        total,
    }))

#[derive(Debug, Deserialize)]
pub struct StockAdjustmentInput {
    pub product_id: Uuid,
    pub batch_id: Uuid,

    // IN, OUT, ADJUSTMENT
    pub movement_type: String,

    // IN / OUT: quantity to add or remove.
    // ADJUSTMENT: target quantity to set.
    pub quantity: Decimal,

    pub reason: String,
}

#[derive(Debug, Serialize, FromRow)]
pub struct StockMovement {
    pub id: Uuid,
    pub product_id: Uuid,
    pub batch_id: Uuid,
    pub movement_type: String,
    pub quantity_change: Decimal,
    pub quantity_before: Decimal,
    pub quantity_after: Decimal,
    pub reason: String,
    pub created_at: DateTime<Utc>,
}

pub async fn create_stock_adjustment(
    State(pool): State<PgPool>,
    Json(input): Json<StockAdjustmentInput>,
) -> Result<(StatusCode, Json<StockMovement>), ApiError> {
    let movement_type = input.movement_type.trim().to_uppercase();
    let reason = input.reason.trim().to_owned();

    if !["IN", "OUT", "ADJUSTMENT"].contains(&movement_type.as_str()) {
        return Err(ApiError::bad_request(
            "movement_type must be IN, OUT or ADJUSTMENT",
        ));
    }

    if input.quantity <= Decimal::ZERO {
        return Err(ApiError::bad_request("Quantity must be greater than zero"));
    }

    if reason.is_empty() {
        return Err(ApiError::bad_request("Reason is required"));
    }

    let mut tx = pool.begin().await.map_err(ApiError::internal)?;

    // Lock the selected batch so simultaneous adjustments
    // cannot overwrite one another.
    let batch = sqlx::query_as::<_, (Uuid, Decimal)>(
        r#"
        SELECT id, quantity_on_hand
        FROM product_batches
        WHERE id = $1
          AND product_id = $2
          AND is_active = TRUE
        FOR UPDATE
        "#,
    )
    .bind(input.batch_id)
    .bind(input.product_id)
    .fetch_optional(&mut *tx)
    .await
    .map_err(ApiError::internal)?;

    let (batch_id, before) = match batch {
        Some(batch) => batch,
        None => {
            return Err(ApiError::not_found(
                "Active batch not found for this product",
            ));
        }
    };

    let after = match movement_type.as_str() {
        "IN" => before + input.quantity,
        "OUT" => before - input.quantity,
        "ADJUSTMENT" => input.quantity,
        _ => unreachable!(),
    };

    if after < Decimal::ZERO {
        return Err(ApiError::conflict("Insufficient stock for this stock-out"));
    }

    let change = after - before;

    if change == Decimal::ZERO {
        return Err(ApiError::bad_request(
            "Adjustment does not change the current stock",
        ));
    }

    sqlx::query(
        r#"
        UPDATE product_batches
        SET quantity_on_hand = $1,
            updated_at = NOW()
        WHERE id = $2
        "#,
    )
    .bind(after)
    .bind(batch_id)
    .execute(&mut *tx)
    .await
    .map_err(ApiError::internal)?;

    let movement = sqlx::query_as::<_, StockMovement>(
        r#"
        INSERT INTO stock_movements (
            product_id,
            batch_id,
            movement_type,
            quantity_change,
            quantity_before,
            quantity_after,
            reason
        )
        VALUES ($1, $2, $3, $4, $5, $6, $7)

        RETURNING
            id,
            product_id,
            batch_id,
            movement_type,
            quantity_change,
            quantity_before,
            quantity_after,
            reason,
            created_at
        "#,
    )
    .bind(input.product_id)
    .bind(batch_id)

pub struct ApiError {
    status: StatusCode,
    message: String,
}

impl ApiError {
    fn bad_request(message: &str) -> Self {
        Self {
            status: StatusCode::BAD_REQUEST,
            message: message.to_owned(),
        }
    }

    fn not_found(message: &str) -> Self {
        Self {
            status: StatusCode::NOT_FOUND,
            message: message.to_owned(),
        }
    }

    fn conflict(message: &str) -> Self {
        Self {
            status: StatusCode::CONFLICT,
            message: message.to_owned(),
        }
    }

    fn internal<E: std::fmt::Display>(error: E) -> Self {
        tracing::error!("Inventory API error: {error}");

        Self {
            status: StatusCode::INTERNAL_SERVER_ERROR,
            message: "Inventory operation failed. Check backend logs.".to_owned(),
        }
    }
}

impl axum::response::IntoResponse for ApiError {
    fn into_response(self) -> axum::response::Response {
        (
            self.status,
            Json(serde_json::json!({
                "error": self.message
            })),
        )
            .into_response()
    }
}

pub async fn list_parties(
    State(pool): State<PgPool>,
    Query(query): Query<PartyQuery>,
) -> Result<Json<PartyList>, (StatusCode, String)> {
    let search = query.search.unwrap_or_default().trim().to_string();

    let role = query.role.unwrap_or_default().trim().to_uppercase();

    if !role.is_empty() && !valid_role(&role) {
        return Err((
            StatusCode::BAD_REQUEST,
            "Role must be SUPPLIER or CUSTOMER".to_string(),
        ));
    }

    let page = query.page.unwrap_or(1).max(1);
    let limit = query.limit.unwrap_or(50).clamp(1, 200);
    let offset = (page - 1).saturating_mul(limit);
    let active = query.active.unwrap_or(true);

    let total: i64 = sqlx::query_scalar(
        r#"
        SELECT COUNT(*)
        FROM parties p
        WHERE
            p.is_active = $1
            AND (
                $2 = ''
                OR p.name ILIKE '%' || $2 || '%'
                OR p.party_code ILIKE '%' || $2 || '%'
                OR COALESCE(p.phone, '') ILIKE '%' || $2 || '%'
                OR COALESCE(p.gstin, '') ILIKE '%' || $2 || '%'
            )
            AND (
                $3 = ''
                OR EXISTS (
                    SELECT 1
                    FROM party_roles r
                    WHERE r.party_id = p.id
                      AND r.role = $3
                )
            )
        "#,
    )
    .bind(active)
    .bind(&search)
    .bind(&role)
    .fetch_one(&pool)
    .await
    .map_err(internal_error)?;

    let data = sqlx::query_as::<_, Party>(
        r#"
        SELECT
            p.id,
            p.party_code,
            p.name,
            p.contact_person,
            p.phone,
            p.alternate_phone,
            p.email,
            p.gstin,
            p.pan,
            p.drug_license_number,
            p.billing_address,
            p.shipping_address,
            p.city,
            p.district,
            p.state,
            p.pincode,
            p.country,
            p.opening_balance,
            p.opening_balance_type,
            p.credit_limit,
            p.credit_days,
            p.notes,
            p.is_active,
            p.created_at,
            p.updated_at,
            COALESCE(
                ARRAY_AGG(pr.role ORDER BY pr.role)
                    FILTER (WHERE pr.role IS NOT NULL),
                ARRAY[]::VARCHAR[]
            ) AS roles
        FROM parties p
        LEFT JOIN party_roles pr ON pr.party_id = p.id
        WHERE
            p.is_active = $1
            AND (
                $2 = ''
                OR p.name ILIKE '%' || $2 || '%'
                OR p.party_code ILIKE '%' || $2 || '%'
                OR COALESCE(p.phone, '') ILIKE '%' || $2 || '%'
                OR COALESCE(p.gstin, '') ILIKE '%' || $2 || '%'
            )
            AND (
                $3 = ''
                OR EXISTS (
                    SELECT 1
                    FROM party_roles r
                    WHERE r.party_id = p.id
                      AND r.role = $3
                )
            )
        GROUP BY p.id
        ORDER BY p.name ASC, p.id ASC
        LIMIT $4 OFFSET $5
        "#,
    )
    .bind(active)
    .bind(&search)
    .bind(&role)
    .bind(limit)
    .bind(offset)
    .fetch_all(&pool)
    .await
    .map_err(internal_error)?;

    Ok(Json(PartyList {

// ============================================================
// INPUT VALIDATION
// ============================================================

fn normalize_input(mut input: PartyInput) -> Result<PartyInput, (StatusCode, String)> {
    input.party_code = input.party_code.trim().to_uppercase();
    input.name = input.name.trim().to_string();

    if input.party_code.is_empty() || input.name.is_empty() {
        return Err((
            StatusCode::BAD_REQUEST,
            "Party code and name are required".to_string(),
        ));
    }

    if input.party_code.len() > 50 || input.name.len() > 250 {
        return Err((
            StatusCode::BAD_REQUEST,
            "Party code or name exceeds the allowed length".to_string(),
        ));
    }

    input.roles = input
        .roles
        .iter()
        .map(|role| role.trim().to_uppercase())
        .collect();

    input.roles.sort();
    input.roles.dedup();

    if input.roles.is_empty() || input.roles.iter().any(|role| !valid_role(role)) {
        return Err((
            StatusCode::BAD_REQUEST,
            "Select at least one valid role: SUPPLIER or CUSTOMER".to_string(),
        ));
    }

    let balance_type = input
        .opening_balance_type
        .as_deref()
        .unwrap_or("PAYABLE")
        .trim()
        .to_uppercase();

    if !matches!(balance_type.as_str(), "PAYABLE" | "RECEIVABLE" | "ADVANCE") {
        return Err((
            StatusCode::BAD_REQUEST,
            "Invalid opening balance type".to_string(),
        ));
    }

    input.opening_balance_type = Some(balance_type);

    let opening = input.opening_balance.unwrap_or(Decimal::ZERO);
    let credit_limit = input.credit_limit.unwrap_or(Decimal::ZERO);
    let credit_days = input.credit_days.unwrap_or(0);

    if opening < Decimal::ZERO || credit_limit < Decimal::ZERO || credit_days < 0 {
        return Err((
            StatusCode::BAD_REQUEST,
            "Balances, credit limit and credit days cannot be negative".to_string(),
        ));
    }

    input.opening_balance = Some(opening);
    input.credit_limit = Some(credit_limit);
    input.credit_days = Some(credit_days);
    input.is_active = Some(input.is_active.unwrap_or(true));

    input.state = Some(
        input
            .state
            .unwrap_or_else(|| "Rajasthan".to_string())
            .trim()
            .to_string(),
    );

    input.pincode = Some(input.pincode.unwrap_or_default().trim().to_string());

    input.country = Some(
        input
            .country
            .unwrap_or_else(|| "India".to_string())
            .trim()
            .to_string(),
    );

    normalize_optional(&mut input.contact_person);
    normalize_optional(&mut input.phone);
    normalize_optional(&mut input.alternate_phone);
    normalize_optional(&mut input.email);
    normalize_optional(&mut input.gstin);
    normalize_optional(&mut input.pan);
    normalize_optional(&mut input.drug_license_number);
    normalize_optional(&mut input.billing_address);
    normalize_optional(&mut input.shipping_address);
    normalize_optional(&mut input.city);
    normalize_optional(&mut input.district);
    normalize_optional(&mut input.notes);

    if let Some(email) = &input.email {
        if !email.contains('@') || email.starts_with('@') || email.ends_with('@') {
            return Err((
                StatusCode::BAD_REQUEST,
                "Email address is invalid".to_string(),
            ));
        }
    }

    if let Some(gstin) = &input.gstin {
        if gstin.len() != 15 || !gstin.chars().all(|c| c.is_ascii_alphanumeric()) {
            return Err((
                StatusCode::BAD_REQUEST,
                "GSTIN must contain 15 alphanumeric characters".to_string(),
            ));
        }
    }

    if let Some(pan) = &input.pan {
        let chars: Vec<char> = pan.chars().collect();

        let valid = chars.len() == 10
            && chars[0..5].iter().all(|c| c.is_ascii_uppercase())
            && chars[5..9].iter().all(|c| c.is_ascii_digit())
            && chars[9].is_ascii_uppercase();

        if !valid {
            return Err((StatusCode::BAD_REQUEST, "PAN format is invalid".to_string()));
        }
    }

    if let Some(pincode) = &input.pincode {
        if !pincode.is_empty()
            && (pincode.len() != 6 || !pincode.chars().all(|c| c.is_ascii_digit()))
        {
            return Err((
                StatusCode::BAD_REQUEST,
                "Pincode must contain 6 digits".to_string(),
            ));
        }
    }

    Ok(input)
}

fn normalize_optional(value: &mut Option<String>) {
    if let Some(text) = value {
        let trimmed = text.trim().to_string();

        *value = if trimmed.is_empty() {
            None
        } else {
            Some(trimmed)
        };
    }
}

fn valid_role(role: &str) -> bool {
    matches!(role, "SUPPLIER" | "CUSTOMER")
}

fn internal_error(error: sqlx::Error) -> (StatusCode, String) {
    tracing::error!("Party database operation failed: {error}");

    (
        StatusCode::INTERNAL_SERVER_ERROR,
        "Unable to process party request".to_string(),
    )
}

fn database_error(error: sqlx::Error) -> (StatusCode, String) {
    if let sqlx::Error::Database(db_error) = &error {
        match db_error.code().as_deref() {
            Some("23505") => {
                return (
                    StatusCode::CONFLICT,
                    "Party code already exists".to_string(),
                );
            }
            Some("23503") => {
                return (
                    StatusCode::CONFLICT,
                    "Party is referenced by another record".to_string(),
                );
            }
            Some("23514") | Some("22P02") => {
                return (
                    StatusCode::BAD_REQUEST,
                    "Party data violates a database constraint".to_string(),
                );
            }
            _ => {}
        }
    }

    internal_error(error)
}

pub async fn list_products(
    State(pool): State<PgPool>,
    Query(query): Query<ProductQuery>,
) -> Result<Json<ProductList>, (StatusCode, Json<ApiError>)> {
    let page = query.page.unwrap_or(1).max(1);
    let limit = query.limit.unwrap_or(50).clamp(1, 100);
    let offset = (page - 1) * limit;
    let search = query.search.unwrap_or_default().trim().to_owned();

    let total: i64 = sqlx::query_scalar(
        r#"
        SELECT COUNT(*)
        FROM public.products p
        LEFT JOIN public.brands b ON b.id = p.brand_id
        WHERE $1 = ''
           OR p.name ILIKE '%' || $1 || '%'
           OR p.sku ILIKE '%' || $1 || '%'
           OR COALESCE(p.generic_name, '') ILIKE '%' || $1 || '%'
           OR COALESCE(b.name, p.brand_name, '') ILIKE '%' || $1 || '%'
           OR EXISTS (
               SELECT 1
               FROM public.product_compositions pc
               JOIN public.compositions c
                   ON c.id = pc.composition_id
               WHERE pc.product_id = p.id
                 AND c.name ILIKE '%' || $1 || '%'
           )
        "#,
    )
    .bind(&search)
    .fetch_one(&pool)
    .await
    .map_err(internal_error)?;

    // Static SQL: no format!() and no dynamically constructed query.
    let sql = r#"
        SELECT
            p.id,
            p.sku,
            p.name,
            p.generic_name,
            p.brand_id,
            COALESCE(b.name, p.brand_name) AS brand_name,

            COALESCE(
                (
                    SELECT array_agg(pc.composition_id ORDER BY c.name)
                    FROM public.product_compositions pc
                    JOIN public.compositions c
                        ON c.id = pc.composition_id
                    WHERE pc.product_id = p.id
                ),
                ARRAY[]::UUID[]
            ) AS composition_ids,

            COALESCE(
                (
                    SELECT array_agg(c.name ORDER BY c.name)
                    FROM public.product_compositions pc
                    JOIN public.compositions c
                        ON c.id = pc.composition_id
                    WHERE pc.product_id = p.id
                ),
                ARRAY[]::TEXT[]
            ) AS composition_names,

            p.dosage_form,
            p.strength,
            p.pack_size,
            p.unit,
            p.hsn_code,
            p.gst_rate,
            p.barcode,
            p.is_prescription_required,
            p.is_active,
            p.schedule_class,
            p.is_narcotic,
            p.units_per_pack

        FROM public.products p
        LEFT JOIN public.brands b ON b.id = p.brand_id

        WHERE $1 = ''
           OR p.name ILIKE '%' || $1 || '%'
           OR p.sku ILIKE '%' || $1 || '%'
           OR COALESCE(p.generic_name, '') ILIKE '%' || $1 || '%'
           OR COALESCE(b.name, p.brand_name, '') ILIKE '%' || $1 || '%'
           OR EXISTS (
               SELECT 1
               FROM public.product_compositions pc
               JOIN public.compositions c
                   ON c.id = pc.composition_id
               WHERE pc.product_id = p.id
                 AND c.name ILIKE '%' || $1 || '%'
           )

        ORDER BY p.name, p.sku
        LIMIT $2 OFFSET $3
    "#;

    let data = sqlx::query_as::<_, Product>(sql)
        .bind(&search)
        .bind(limit)
        .bind(offset)
        .fetch_all(&pool)
        .await
        .map_err(internal_error)?;

    Ok(Json(ProductList {
        data,
        page,
        limit,
        total,
    }))
}

            $10,
            COALESCE($11, 0),
            $12,

// Composition relationship helpers
// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

fn validate_composition_ids(ids: &[Uuid]) -> Result<(), (StatusCode, Json<ApiError>)> {
    let mut unique = std::collections::HashSet::new();

    for id in ids {
        if !unique.insert(*id) {
            return Err(bad_request("Duplicate composition IDs are not allowed"));
        }
    }

    Ok(())
}

async fn replace_compositions(
    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
    product_id: Uuid,
    composition_ids: &[Uuid],
) -> Result<(), (StatusCode, Json<ApiError>)> {
    sqlx::query("DELETE FROM public.product_compositions WHERE product_id = $1")
        .bind(product_id)
        .execute(&mut **tx)
        .await
        .map_err(write_error)?;

    for composition_id in composition_ids {
        sqlx::query(
            r#"
            INSERT INTO public.product_compositions (
                product_id,
                composition_id
            )
            VALUES ($1, $2)
            "#,
        )
        .bind(product_id)
        .bind(composition_id)
        .execute(&mut **tx)
        .await
        .map_err(write_error)?;
    }

    Ok(())
}

// DELETE /api/products/{id}
// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

pub async fn delete_product(
    State(pool): State<PgPool>,
    Path(id): Path<Uuid>,
) -> Result<StatusCode, (StatusCode, Json<ApiError>)> {
    let result = sqlx::query("DELETE FROM public.products WHERE id = $1")
        .bind(id)
        .execute(&pool)
        .await
        .map_err(write_error)?;

    if result.rows_affected() == 0 {
        return Err(not_found());
    }

    Ok(StatusCode::NO_CONTENT)
}

// Validation
// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

fn normalized_schedule(value: Option<&str>) -> String {
    value.unwrap_or("OTC").trim().to_uppercase()
}

fn validate_product(
    sku: &str,
    name: &str,
    gst_rate: Option<Decimal>,
    schedule_class: Option<&str>,
    units_per_pack: Option<Decimal>,
) -> Result<(), (StatusCode, Json<ApiError>)> {
    if sku.is_empty() || name.is_empty() {
        return Err(bad_request("SKU and product name are required"));
    }

    if sku.len() > 80 || name.len() > 250 {
        return Err(bad_request(
            "SKU or product name exceeds the allowed length",
        ));
    }

    if let Some(rate) = gst_rate {
        if rate < Decimal::ZERO || rate > Decimal::from(100) {
            return Err(bad_request("GST rate must be between 0 and 100"));
        }
    }

    if let Some(class) = schedule_class {
        let class = class.trim().to_uppercase();

        if !["OTC", "H", "H1", "X"].contains(&class.as_str()) {
            return Err(bad_request("Schedule class must be OTC, H, H1 or X"));
        }
    }

    if let Some(units) = units_per_pack {
        if units <= Decimal::ZERO {
            return Err(bad_request("Units per pack must be greater than zero"));
        }
    }

    Ok(())
}

// Error responses
// â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

fn bad_request(message: &str) -> (StatusCode, Json<ApiError>) {
    (
        StatusCode::BAD_REQUEST,
        Json(ApiError {
            error: message.to_owned(),
        }),
    )
}

fn not_found() -> (StatusCode, Json<ApiError>) {
    (
        StatusCode::NOT_FOUND,
        Json(ApiError {
            error: "Product not found".to_owned(),
        }),
    )
}

fn internal_error(error: sqlx::Error) -> (StatusCode, Json<ApiError>) {
    tracing::error!("Product query failed: {error}");

    (
        StatusCode::INTERNAL_SERVER_ERROR,
        Json(ApiError {
            error: "Database query failed".to_owned(),
        }),
    )
}

fn write_error(error: sqlx::Error) -> (StatusCode, Json<ApiError>) {
    if let sqlx::Error::Database(ref db_error) = error {
        if db_error.is_unique_violation() {
            return (
                StatusCode::CONFLICT,
                Json(ApiError {
                    error: "SKU or another unique field already exists".to_owned(),
                }),
            );
        }

        if db_error.is_foreign_key_violation() {
            return (
                StatusCode::CONFLICT,
                Json(ApiError {
                    error: "A referenced brand, composition or related record does not exist, or the product is referenced elsewhere".to_owned(),
                }),
            );
        }

        if db_error.is_check_violation() {
            return (
                StatusCode::BAD_REQUEST,
                Json(ApiError {
                    error: "One or more product values violate a database constraint".to_owned(),
                }),
            );
        }
    }

    tracing::error!("Product write failed: {error}");

    (
        StatusCode::INTERNAL_SERVER_ERROR,
        Json(ApiError {
            error: "Product operation failed".to_owned(),
        }),
    )
}
