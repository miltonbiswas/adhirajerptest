

========== 0001_initial_schema.sql ==========
BEGIN;

CREATE TABLE categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(120) NOT NULL,
    description TEXT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_categories_name UNIQUE (name)
);

CREATE TABLE manufacturers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    contact_person VARCHAR(150),
    phone VARCHAR(20),
    email VARCHAR(254),
    address TEXT,
    gstin VARCHAR(15),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_manufacturers_name UNIQUE (name)
);

CREATE TABLE products (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku VARCHAR(80) NOT NULL,
    name VARCHAR(250) NOT NULL,
    generic_name VARCHAR(250),
    category_id UUID REFERENCES categories(id) ON DELETE RESTRICT,
    manufacturer_id UUID REFERENCES manufacturers(id) ON DELETE RESTRICT,
    brand_name VARCHAR(150),
    dosage_form VARCHAR(80),
    strength VARCHAR(80),
    pack_size VARCHAR(80),
    unit VARCHAR(30) NOT NULL DEFAULT 'PCS',
    hsn_code VARCHAR(8),
    gst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    barcode VARCHAR(100),
    is_prescription_required BOOLEAN NOT NULL DEFAULT FALSE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_products_sku UNIQUE (sku),
    CONSTRAINT chk_products_gst_rate
        CHECK (gst_rate >= 0 AND gst_rate <= 100)
);

CREATE INDEX idx_products_name ON products (name);
CREATE INDEX idx_products_generic_name ON products (generic_name);
CREATE INDEX idx_products_category_id ON products (category_id);
CREATE INDEX idx_products_manufacturer_id ON products (manufacturer_id);
CREATE INDEX idx_products_barcode ON products (barcode)
    WHERE barcode IS NOT NULL;

CREATE TABLE product_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    product_id UUID NOT NULL REFERENCES products(id) ON DELETE RESTRICT,
    batch_number VARCHAR(100) NOT NULL,
    manufacturing_date DATE,
    expiry_date DATE,
    purchase_rate NUMERIC(14,4) NOT NULL DEFAULT 0,
    mrp NUMERIC(14,4) NOT NULL DEFAULT 0,
    selling_rate NUMERIC(14,4) NOT NULL DEFAULT 0,
    quantity_on_hand NUMERIC(14,3) NOT NULL DEFAULT 0,
    reorder_level NUMERIC(14,3) NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_product_batch UNIQUE (product_id, batch_number),
    CONSTRAINT chk_batch_purchase_rate CHECK (purchase_rate >= 0),
    CONSTRAINT chk_batch_mrp CHECK (mrp >= 0),
    CONSTRAINT chk_batch_selling_rate CHECK (selling_rate >= 0),
    CONSTRAINT chk_batch_quantity CHECK (quantity_on_hand >= 0),
    CONSTRAINT chk_batch_reorder_level CHECK (reorder_level >= 0),
    CONSTRAINT chk_batch_dates CHECK (
        manufacturing_date IS NULL
        OR expiry_date IS NULL
        OR manufacturing_date <= expiry_date
    )
);

CREATE INDEX idx_batches_product_id ON product_batches (product_id);

CREATE INDEX idx_batches_fefo
    ON product_batches (expiry_date, product_id)
    WHERE is_active = TRUE;

CREATE TABLE suppliers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    supplier_code VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(250) NOT NULL,
    contact_person VARCHAR(150),
    phone VARCHAR(20),
    email VARCHAR(254),
    gstin VARCHAR(15),
    drug_license_number VARCHAR(100),
    address TEXT,
    opening_balance NUMERIC(14,2) NOT NULL DEFAULT 0,
    opening_balance_type VARCHAR(10) NOT NULL DEFAULT 'PAYABLE',
    credit_days INTEGER NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT chk_supplier_opening_balance_type
        CHECK (opening_balance_type IN ('PAYABLE', 'ADVANCE')),
    CONSTRAINT chk_supplier_credit_days CHECK (credit_days >= 0)
);

CREATE INDEX idx_suppliers_name ON suppliers (name);
CREATE INDEX idx_suppliers_phone ON suppliers (phone);

CREATE TABLE customers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    customer_code VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(250) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(254),
    gstin VARCHAR(15),
    address TEXT,
    opening_balance NUMERIC(14,2) NOT NULL DEFAULT 0,
    opening_balance_type VARCHAR(10) NOT NULL DEFAULT 'RECEIVABLE',
    credit_limit NUMERIC(14,2) NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT chk_customer_opening_balance_type
        CHECK (opening_balance_type IN ('RECEIVABLE', 'ADVANCE')),
    CONSTRAINT chk_customer_credit_limit CHECK (credit_limit >= 0)
);

CREATE INDEX idx_customers_name ON customers (name);
CREATE INDEX idx_customers_phone ON customers (phone);

COMMIT;


========== 0002_product_brand_composition.sql ==========
BEGIN;

-- ============================================
-- 1. BRANDS MASTER
-- ============================================

CREATE TABLE brands (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name TEXT NOT NULL,
    normalized_name TEXT GENERATED ALWAYS AS (
        lower(regexp_replace(trim(name), '\s+', ' ', 'g'))
    ) STORED,

    manufacturer_id UUID REFERENCES manufacturers(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    description TEXT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT brands_name_not_empty
        CHECK (length(trim(name)) > 0)
);

CREATE UNIQUE INDEX uq_brands_normalized_name
    ON brands (normalized_name);

CREATE INDEX idx_brands_name
    ON brands (name);

CREATE INDEX idx_brands_manufacturer_id
    ON brands (manufacturer_id);


-- ============================================
-- 2. COMPOSITIONS MASTER
-- ============================================

CREATE TABLE compositions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name TEXT NOT NULL,
    normalized_name TEXT GENERATED ALWAYS AS (
        lower(regexp_replace(trim(name), '\s+', ' ', 'g'))
    ) STORED,

    description TEXT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT compositions_name_not_empty
        CHECK (length(trim(name)) > 0)
);

CREATE UNIQUE INDEX uq_compositions_normalized_name
    ON compositions (normalized_name);

CREATE INDEX idx_compositions_name
    ON compositions (name);


-- ============================================
-- 3. PRODUCT-COMPOSITION MAPPING
-- Supports multiple ingredients per product
-- ============================================

CREATE TABLE product_compositions (
    product_id UUID NOT NULL
        REFERENCES products(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    composition_id UUID NOT NULL
        REFERENCES compositions(id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    strength TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    PRIMARY KEY (product_id, composition_id)
);

CREATE INDEX idx_product_compositions_composition_id
    ON product_compositions (composition_id);


COMMIT;



========== 0003_product_brand_link.sql ==========
BEGIN;

-- Link each product to its brand master record.
ALTER TABLE products
ADD COLUMN brand_id UUID;

-- Prevent deleting a brand while products reference it.
ALTER TABLE products
ADD CONSTRAINT fk_products_brand
FOREIGN KEY (brand_id)
REFERENCES brands(id)
ON UPDATE RESTRICT
ON DELETE RESTRICT;

-- Speed up brand-based product searches.
CREATE INDEX idx_products_brand_id
ON products (brand_id);

COMMIT;


========== 0004_product_import_staging.sql ==========
CREATE TABLE product_import_staging (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

    source_file TEXT NOT NULL DEFAULT 'prodd.xlsx',
    source_row INTEGER NOT NULL,

    original_name TEXT NOT NULL,
    cleaned_name TEXT NOT NULL,
    normalized_name TEXT NOT NULL,

    proposed_sku VARCHAR(50) NOT NULL,
    review_status VARCHAR(30) NOT NULL DEFAULT 'PENDING_REVIEW',

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_product_import_staging_source_row
        UNIQUE (source_file, source_row),

    CONSTRAINT uq_product_import_staging_proposed_sku
        UNIQUE (proposed_sku),

    CONSTRAINT chk_product_import_staging_names
        CHECK (
            LENGTH(BTRIM(original_name)) > 0
            AND LENGTH(BTRIM(cleaned_name)) > 0
            AND LENGTH(BTRIM(normalized_name)) > 0
        ),

    CONSTRAINT chk_product_import_staging_review_status
        CHECK (
            review_status IN (
                'PENDING_REVIEW',
                'APPROVED',
                'REJECTED'
            )
        )
);

CREATE INDEX idx_product_import_staging_normalized_name
    ON product_import_staging (normalized_name);

CREATE INDEX idx_product_import_staging_review_status
    ON product_import_staging (review_status);


========== 0005_inventory_movements.sql ==========
CREATE TABLE IF NOT EXISTS stock_movements (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    product_id UUID NOT NULL
        REFERENCES products(id) ON DELETE RESTRICT,

    batch_id UUID NOT NULL
        REFERENCES product_batches(id) ON DELETE RESTRICT,

    movement_type VARCHAR(20) NOT NULL
        CHECK (movement_type IN ('IN', 'OUT', 'ADJUSTMENT')),

    quantity_change NUMERIC(14,3) NOT NULL,
    quantity_before NUMERIC(14,3) NOT NULL,
    quantity_after NUMERIC(14,3) NOT NULL,

    reason TEXT NOT NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_stock_movement_quantities
        CHECK (
            quantity_change <> 0
            AND quantity_before >= 0
            AND quantity_after >= 0
        )
);

CREATE INDEX IF NOT EXISTS idx_stock_movements_product
    ON stock_movements (product_id, created_at DESC);

CREATE INDEX IF NOT EXISTS idx_stock_movements_batch
    ON stock_movements (batch_id, created_at DESC);

========== 0007_product_classification_pack.sql ==========
BEGIN;

ALTER TABLE products
    ADD COLUMN IF NOT EXISTS schedule_class VARCHAR(10)
        NOT NULL DEFAULT 'OTC',
    ADD COLUMN IF NOT EXISTS is_narcotic BOOLEAN
        NOT NULL DEFAULT FALSE,
    ADD COLUMN IF NOT EXISTS units_per_pack NUMERIC(14,3)
        NOT NULL DEFAULT 1;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_constraint
        WHERE conname = 'chk_products_schedule_class'
          AND conrelid = 'products'::regclass
    ) THEN
        ALTER TABLE products
            ADD CONSTRAINT chk_products_schedule_class
            CHECK (schedule_class IN ('OTC', 'H', 'H1', 'X'));
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_constraint
        WHERE conname = 'chk_products_units_per_pack'
          AND conrelid = 'products'::regclass
    ) THEN
        ALTER TABLE products
            ADD CONSTRAINT chk_products_units_per_pack
            CHECK (units_per_pack > 0);
    END IF;
END
$$;

CREATE INDEX IF NOT EXISTS idx_products_schedule_class
    ON products (schedule_class);

COMMIT;


========== 0008_party_master.sql ==========
BEGIN;

-- Common party master
CREATE TABLE parties (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    party_code VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(250) NOT NULL,
    contact_person VARCHAR(150),

    phone VARCHAR(20),
    alternate_phone VARCHAR(20),
    email VARCHAR(254),

    gstin VARCHAR(15),
    pan VARCHAR(10),
    drug_license_number VARCHAR(100),

    billing_address TEXT,
    shipping_address TEXT,
    city VARCHAR(100),
    district VARCHAR(100),
    state VARCHAR(100) NOT NULL DEFAULT 'Rajasthan',
    pincode VARCHAR(6) NOT NULL DEFAULT '',
    country VARCHAR(100) NOT NULL DEFAULT 'India',

    opening_balance NUMERIC(14,2) NOT NULL DEFAULT 0,
    opening_balance_type VARCHAR(12) NOT NULL DEFAULT 'PAYABLE',

    credit_limit NUMERIC(14,2) NOT NULL DEFAULT 0,
    credit_days INTEGER NOT NULL DEFAULT 0,

    notes TEXT,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_parties_opening_balance_type
        CHECK (
            opening_balance_type IN (
                'PAYABLE',
                'RECEIVABLE',
                'ADVANCE'
            )
        ),

    CONSTRAINT chk_parties_credit_limit
        CHECK (credit_limit >= 0),

    CONSTRAINT chk_parties_credit_days
        CHECK (credit_days >= 0),

    CONSTRAINT chk_parties_pincode
        CHECK (pincode = '' OR pincode ~ '^[0-9]{6}$'),

    CONSTRAINT chk_parties_gstin
        CHECK (gstin IS NULL OR gstin = '' OR gstin ~ '^[0-9A-Z]{15}$'),

    CONSTRAINT chk_parties_pan
        CHECK (pan IS NULL OR pan = '' OR pan ~ '^[A-Z]{5}[0-9]{4}[A-Z]$')
);

-- A party can have one or both roles.
CREATE TABLE party_roles (
    party_id UUID NOT NULL
        REFERENCES parties(id) ON DELETE RESTRICT,

    role VARCHAR(10) NOT NULL,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT pk_party_roles PRIMARY KEY (party_id, role),

    CONSTRAINT chk_party_roles_role
        CHECK (role IN ('SUPPLIER', 'CUSTOMER'))
);

-- Search and filtering indexes.
CREATE INDEX idx_parties_name
    ON parties (name);

CREATE INDEX idx_parties_phone
    ON parties (phone)
    WHERE phone IS NOT NULL;

CREATE INDEX idx_parties_gstin
    ON parties (gstin)
    WHERE gstin IS NOT NULL AND gstin <> '';

CREATE INDEX idx_party_roles_role
    ON party_roles (role, party_id);

COMMIT;


========== 0009_purchase_module.sql ==========
BEGIN;

-- =========================================================
-- PURCHASE INVOICES
-- =========================================================

CREATE TABLE purchases (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    purchase_number VARCHAR(50) NOT NULL UNIQUE,

    supplier_id UUID NOT NULL
        REFERENCES parties(id) ON DELETE RESTRICT,

    supplier_invoice_number VARCHAR(100) NOT NULL,
    invoice_date DATE NOT NULL,
    due_date DATE,

    status VARCHAR(12) NOT NULL DEFAULT 'DRAFT'
        CHECK (status IN ('DRAFT', 'POSTED', 'CANCELLED')),

    subtotal NUMERIC(16,2) NOT NULL DEFAULT 0,
    discount_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    taxable_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    tax_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    other_charges NUMERIC(16,2) NOT NULL DEFAULT 0,
    round_off NUMERIC(16,2) NOT NULL DEFAULT 0,
    grand_total NUMERIC(16,2) NOT NULL DEFAULT 0,

    paid_amount NUMERIC(16,2) NOT NULL DEFAULT 0,

    notes TEXT,

    posted_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_purchase_amounts CHECK (
        subtotal >= 0
        AND discount_amount >= 0
        AND taxable_amount >= 0
        AND tax_amount >= 0
        AND other_charges >= 0
        AND grand_total >= 0
        AND paid_amount >= 0
        AND paid_amount <= grand_total
    )
);

-- Duplicate supplier invoice prevention.
CREATE UNIQUE INDEX uq_purchase_supplier_invoice
    ON purchases (
        supplier_id,
        LOWER(BTRIM(supplier_invoice_number))
    )
    WHERE status <> 'CANCELLED';

CREATE INDEX idx_purchases_supplier_date
    ON purchases (supplier_id, invoice_date DESC);

CREATE INDEX idx_purchases_status_date
    ON purchases (status, invoice_date DESC);


-- =========================================================
-- PURCHASE LINE ITEMS
-- =========================================================

CREATE TABLE purchase_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    purchase_id UUID NOT NULL
        REFERENCES purchases(id) ON DELETE RESTRICT,

    line_number INTEGER NOT NULL,

    product_id UUID NOT NULL
        REFERENCES products(id) ON DELETE RESTRICT,

    batch_id UUID
        REFERENCES product_batches(id) ON DELETE RESTRICT,

    batch_number VARCHAR(100) NOT NULL,
    manufacturing_date DATE,
    expiry_date DATE,

    quantity NUMERIC(14,3) NOT NULL,
    free_quantity NUMERIC(14,3) NOT NULL DEFAULT 0,

    purchase_rate NUMERIC(14,4) NOT NULL,
    mrp NUMERIC(14,4) NOT NULL,
    selling_rate NUMERIC(14,4) NOT NULL DEFAULT 0,

    discount_percent NUMERIC(5,2) NOT NULL DEFAULT 0,
    discount_amount NUMERIC(16,2) NOT NULL DEFAULT 0,

    gst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    taxable_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    cgst_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    sgst_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    igst_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    total_amount NUMERIC(16,2) NOT NULL DEFAULT 0,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_purchase_item_line
        UNIQUE (purchase_id, line_number),

    CONSTRAINT chk_purchase_item_quantity
        CHECK (quantity > 0 AND free_quantity >= 0),

    CONSTRAINT chk_purchase_item_rates
        CHECK (
            purchase_rate >= 0
            AND mrp >= 0
            AND selling_rate >= 0
        ),

    CONSTRAINT chk_purchase_item_discount
        CHECK (
            discount_percent BETWEEN 0 AND 100
            AND discount_amount >= 0
        ),

    CONSTRAINT chk_purchase_item_gst
        CHECK (gst_rate BETWEEN 0 AND 100),

    CONSTRAINT chk_purchase_item_amounts
        CHECK (
            taxable_amount >= 0
            AND cgst_amount >= 0
            AND sgst_amount >= 0
            AND igst_amount >= 0
            AND total_amount >= 0
        ),

    CONSTRAINT chk_purchase_item_dates
        CHECK (
            manufacturing_date IS NULL
            OR expiry_date IS NULL
            OR manufacturing_date <= expiry_date
        )
);

CREATE INDEX idx_purchase_items_purchase
    ON purchase_items (purchase_id, line_number);

CREATE INDEX idx_purchase_items_product
    ON purchase_items (product_id);


-- =========================================================
-- SUPPLIER LEDGER
-- =========================================================

CREATE TABLE supplier_ledger_entries (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    supplier_id UUID NOT NULL
        REFERENCES parties(id) ON DELETE RESTRICT,

    purchase_id UUID
        REFERENCES purchases(id) ON DELETE RESTRICT,

    entry_type VARCHAR(20) NOT NULL
        CHECK (
            entry_type IN (
                'PURCHASE',
                'PAYMENT',
                'CREDIT_NOTE',
                'DEBIT_NOTE',
                'ADJUSTMENT'
            )
        ),

    debit_amount NUMERIC(16,2) NOT NULL DEFAULT 0,
    credit_amount NUMERIC(16,2) NOT NULL DEFAULT 0,

    reference_number VARCHAR(100),
    notes TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_supplier_ledger_amounts CHECK (
        debit_amount >= 0
        AND credit_amount >= 0
        AND NOT (
            debit_amount > 0
            AND credit_amount > 0
        )
        AND (
            debit_amount > 0
            OR credit_amount > 0
        )
    )
);

CREATE INDEX idx_supplier_ledger_supplier_date
    ON supplier_ledger_entries (supplier_id, created_at DESC);

CREATE UNIQUE INDEX uq_supplier_ledger_purchase
    ON supplier_ledger_entries (purchase_id)
    WHERE entry_type = 'PURCHASE';


COMMIT;


========== 0010_hsn_master.sql ==========
CREATE TABLE IF NOT EXISTS hsn_master (
    hsn_code VARCHAR(8) PRIMARY KEY,
    description TEXT NOT NULL DEFAULT '',
    gst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    cgst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    sgst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    igst_rate NUMERIC(5,2) NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT hsn_master_gst_rate_check
        CHECK (gst_rate >= 0 AND gst_rate <= 100),

    CONSTRAINT hsn_master_cgst_rate_check
        CHECK (cgst_rate >= 0 AND cgst_rate <= 100),

    CONSTRAINT hsn_master_sgst_rate_check
        CHECK (sgst_rate >= 0 AND sgst_rate <= 100),

    CONSTRAINT hsn_master_igst_rate_check
        CHECK (igst_rate >= 0 AND igst_rate <= 100)
);

CREATE INDEX IF NOT EXISTS idx_hsn_master_active
    ON hsn_master (hsn_code)
    WHERE is_active = TRUE;
